2. Create test file tests/test_rbac_enforcement.py:
   ```python
   import pytest
   from fastapi.testclient import TestClient
   from src.api.main import app
   from src.rbac.models import Role, Permission, UserRole
   from src.auth.models import User

   client = TestClient(app)

   def test_unauthenticated_user_denied(test_client):
       """Unauthenticated requests should return 401"""
       response = test_client.get("/api/findings")
       assert response.status_code == 401

   def test_user_without_role_denied(test_client, test_user_no_role):
       """User without role should return 403"""
       response = test_client.get(
           "/api/findings",
           headers={"Authorization": f"Bearer {test_user_no_role.token}"}
       )
       assert response.status_code == 403
       assert "Permission denied" in response.json()["detail"]

   def test_analyst_can_read_findings(test_client, test_user_analyst):
       """Analyst role should access findings:read endpoint"""
       response = test_client.get(
           "/api/findings",
           headers={"Authorization": f"Bearer {test_user_analyst.token}"}
       )
       assert response.status_code == 200

   def test_analyst_cannot_delete_findings(test_client, test_user_analyst):
       """Analyst role should NOT access findings:delete endpoint"""
       response = test_client.delete(
           "/api/findings/test-finding-id",
           headers={"Authorization": f"Bearer {test_user_analyst.token}"}
       )
       assert response.status_code == 403

   def test_super_admin_wildcard_access(test_client, test_user_super_admin):
       """Super admin (*:*) should access all endpoints"""
       endpoints = [
           "/api/findings",
           "/api/scans",
           "/api/repositories",
           "/api/organizations",
       ]
       for endpoint in endpoints:
           response = test_client.get(
               endpoint,
               headers={"Authorization": f"Bearer {test_user_super_admin.token}"}
           )
           assert response.status_code in [200, 404], f"Failed on {endpoint}"

   def test_role_hierarchy(test_client, test_user_admin, test_user_manager):
       """Admin (level 2) should access Manager (level 4) resources"""
       # Admin can access manager-level resources
       response = test_client.get(
           "/api/reports",
           headers={"Authorization": f"Bearer {test_user_admin.token}"}
       )
       assert response.status_code == 200

       # Manager cannot access admin-level resources
       response = test_client.post(
           "/api/organizations",
           headers={"Authorization": f"Bearer {test_user_manager.token}"},
           json={"name": "test"}
       )
       assert response.status_code == 403

   def test_audit_log_on_authorization_failure(test_client, test_user_no_role, caplog):
       """Authorization failures should be logged"""
       response = test_client.get(
           "/api/findings",
           headers={"Authorization": f"Bearer {test_user_no_role.token}"}
       )
       assert response.status_code == 403
       assert "authorization.denied" in caplog.text
       assert "findings:read" in caplog.text
   ```

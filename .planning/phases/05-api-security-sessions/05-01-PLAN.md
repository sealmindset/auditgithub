Create `src/auth/tokens.py` with refresh token management:

1. **generate_refresh_token(user_sub: str, tenant_id: str) -> str:**
   - Create signed JWT with claims: {"sub": user_sub, "tenant_id": tenant_id, "type": "refresh", "exp": 7 days, "jti": uuid4()}
   - Sign with HS256 using SECRET_KEY from environment (not RS256 - we control these tokens)
   - Store jti (JWT ID) in Redis with TTL=7 days for rotation tracking
   - Return token string

2. **rotate_refresh_token(old_token: str) -> tuple[str, dict]:**
   - Validate old token signature and expiry
   - Check jti is NOT in blacklist (Redis key: "blacklist:{jti}")
   - Check jti EXISTS in rotation tracker (Redis key: "refresh:{jti}") - prevents reuse
   - Extract user_sub and tenant_id from old token
   - DELETE old jti from rotation tracker (one-time use)
   - Generate NEW refresh token with generate_refresh_token()
   - Return (new_token, user_claims)

3. **revoke_token(jti: str, exp: int):**
   - Add jti to blacklist (Redis SET with TTL = exp - now)
   - Remove from rotation tracker if exists

class RefreshResponse(BaseModel):
    refresh_token: str
    access_token: str  # We'll add access token generation in this task
    token_type: str = "bearer"

@router.post("/refresh", response_model=RefreshResponse)
async def refresh_tokens(request: RefreshRequest):
    """
    Refresh access and refresh tokens.

    Security:
    - Rotates refresh token (one-time use)
    - Validates old token before issuing new one
    - Prevents token reuse attacks

    try:
        new_refresh_token, user_claims = rotate_refresh_token(request.refresh_token)

        # Generate new access token (short-lived)

            user_sub=user_claims["sub"],
            tenant_id=user_claims["tenant_id"],
            email=user_claims.get("email", ""),
            name=user_claims.get("name", "")
        )

            refresh_token=new_refresh_token,
            access_token=access_token

@router.post("/revoke", status_code=status.HTTP_204_NO_CONTENT)
async def revoke_current_token(
    request: Request,
    current_user: User = Depends(get_current_user)
):
    """
    Revoke the current user's access token.

    Adds token JTI to blacklist until natural expiry.
    Useful for logout or security incidents.
    """
    from src.auth.tokens import revoke_token
    from src.auth.dependencies import get_token_jti

    try:
        jti = get_token_jti(request)
        exp = request.state.token_exp  # Also store exp in dependencies.py
        revoke_token(jti, exp)

        return None  # 204 No Content
    except Exception as e:
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail=f"Failed to revoke token: {str(e)}"
        )

   def revoke_token(jti: str, exp: int):
       """
       Add token JTI to blacklist.

       Args:
           jti: Token JTI (unique identifier)
           exp: Token expiry timestamp (Unix time)
       """
       from datetime import datetime

       ttl = exp - int(datetime.utcnow().timestamp())
       if ttl > 0:
           redis_client.setex(f"blacklist:{jti}", ttl, "1")

**Current State:**
- Cookie-based sessions exist (via Starlette SessionMiddleware)
- No session expiry tracking - sessions persist indefinitely
- No idle timeout - inactive users stay logged in forever
- No last_activity tracking - can't distinguish active from idle sessions
- No session cleanup - memory/storage leak over time

**Security Requirements:**
- Absolute session timeout: 8 hours (session expires regardless of activity)
- Idle timeout: 30 minutes (session expires after 30 min of inactivity)
- Activity tracking: Update last_activity on every authenticated request
- Session cleanup: Remove expired sessions from storage every 5 minutes
- Graceful expiry: Return 401 with clear error message, not silent failure

**Tech Stack Available:**
- Redis (for session metadata storage)
- Starlette SessionMiddleware (current session backend)
- FastAPI BackgroundTasks (for periodic cleanup)
- APScheduler (alternative for periodic jobs) - add if needed

**Architecture Decision:**
- Keep Starlette SessionMiddleware for session cookie management
- Add Redis-backed metadata tracking (created_at, last_activity, user_sub)
- Middleware checks metadata on every request and enforces timeouts
- Background job cleans up expired sessions periodically

**Scope Boundaries:**
- Do NOT migrate to Redis sessions entirely (Starlette SessionMiddleware works fine)
- Do NOT add "remember me" functionality (out of scope)
- Focus: Expiry enforcement, activity tracking, cleanup

1. **SessionMetadata model:**
   ```python
   from pydantic import BaseModel
   from datetime import datetime

   class SessionMetadata(BaseModel):
       user_sub: str
       tenant_id: str
       created_at: datetime
       last_activity: datetime
       provider: str

       def is_expired(self, absolute_timeout_hours: int, idle_timeout_minutes: int) -> tuple[bool, str]:
           """
           Check if session is expired.

           Returns:
               (is_expired, reason) - reason is "absolute" or "idle" or None
           """
           now = datetime.utcnow()

           # Check absolute timeout (e.g., 8 hours since creation)
           age_hours = (now - self.created_at).total_seconds() / 3600
           if age_hours > absolute_timeout_hours:
               return (True, "absolute")

           # Check idle timeout (e.g., 30 minutes since last activity)
           idle_minutes = (now - self.last_activity).total_seconds() / 60
           if idle_minutes > idle_timeout_minutes:
               return (True, "idle")

           return (False, None)
   ```

2. **Session storage functions (Redis-backed):**

   import json
   from src.auth.tokens import redis_client  # Reuse Redis client from Plan 05-01

   def get_session_metadata(session_id: str) -> SessionMetadata | None:
       """Fetch session metadata from Redis."""
       data = redis_client.get(f"session:{session_id}")
       if data:
           return SessionMetadata(**json.loads(data))
       return None

   def set_session_metadata(session_id: str, metadata: SessionMetadata, ttl_hours: int):
       """Store session metadata in Redis with TTL."""
       redis_client.setex(
           f"session:{session_id}",
           ttl_hours * 3600,  # Convert hours to seconds
           json.dumps(metadata.dict(), default=str)
       )

   def update_last_activity(session_id: str):
       """Update last_activity timestamp for session."""
       metadata = get_session_metadata(session_id)
       if metadata:
           metadata.last_activity = datetime.utcnow()
           from src.auth.config import settings
           set_session_metadata(session_id, metadata, settings.SESSION_ABSOLUTE_TIMEOUT_HOURS)

   def delete_session(session_id: str):
       """Delete session metadata from Redis."""
       redis_client.delete(f"session:{session_id}")
   ```

async def get_current_user_from_session(request: Request) -> User:
    """
    Extract current user from session (cookie-based auth).

    Enforces session expiry based on absolute and idle timeouts.
    """
    user_data = request.session.get('user')
    if not user_data:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Not authenticated"
        )

    # Check session metadata for expiry
    session_id = request.cookies.get('session')  # Starlette session cookie name
    if session_id:
        from src.auth.session import get_session_metadata, update_last_activity
        from src.auth.config import settings

        metadata = get_session_metadata(session_id)

        if not metadata:
            # Session metadata missing - treat as expired
            request.session.clear()
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Session expired (metadata not found)"
            )

        # Check expiry
        is_expired, reason = metadata.is_expired(
            settings.SESSION_ABSOLUTE_TIMEOUT_HOURS,
            settings.SESSION_IDLE_TIMEOUT_MINUTES
        )

        if is_expired:
            # Clear session and raise 401
            request.session.clear()
            from src.auth.session import delete_session
            delete_session(session_id)

            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail=f"Session expired ({reason} timeout)"
            )

        # Update last activity (async to avoid blocking request)
        update_last_activity(session_id)

    return User(**user_data)
```

```python
from starlette.middleware.base import BaseHTTPMiddleware
from fastapi import Request
import logging

logger = logging.getLogger(__name__)

class SessionActivityMiddleware(BaseHTTPMiddleware):
    """
    Middleware to track session activity on every authenticated request.

    Updates last_activity timestamp in Redis for idle timeout enforcement.
    """

    async def dispatch(self, request: Request, call_next):
        # Process request
        response = await call_next(request)

        # Update activity timestamp if user is authenticated
        session_id = request.cookies.get('session')
        user_data = request.session.get('user')

        if session_id and user_data:
            # Update last_activity in Redis (non-blocking)
            from src.auth.session import update_last_activity
            try:
                update_last_activity(session_id)
            except Exception as e:
                logger.warning(f"Failed to update session activity: {e}")

        return response
```

Create periodic cleanup job in `src/auth/cleanup.py`:

```python
#!/usr/bin/env python3
"""
Background job to clean up expired sessions from Redis.

Runs every 5 minutes and removes session metadata for expired sessions.
This prevents Redis memory leaks from abandoned sessions.
"""

import schedule
import time
import logging
from datetime import datetime
from src.auth.session import redis_client
from src.auth.config import settings

logger = logging.getLogger(__name__)

def cleanup_expired_sessions():
    """
    Scan Redis for expired sessions and delete them.

    Uses SCAN cursor to avoid blocking Redis on large datasets.
    """
    logger.info("Running session cleanup job...")

    cleaned_count = 0
    cursor = 0

    while True:
        # SCAN for session keys (cursor-based to avoid blocking)
        cursor, keys = redis_client.scan(
            cursor=cursor,
            match="session:*",
            count=100  # Process 100 keys per iteration
        )

        for key in keys:
            try:
                # Check if key still exists (may have auto-expired via TTL)
                if not redis_client.exists(key):
                    continue

                # Fetch metadata
                import json
                from src.auth.session import SessionMetadata
                data = redis_client.get(key)
                if not data:
                    continue

                metadata = SessionMetadata(**json.loads(data))

                # Check if expired
                is_expired, reason = metadata.is_expired(
                    settings.SESSION_ABSOLUTE_TIMEOUT_HOURS,
                    settings.SESSION_IDLE_TIMEOUT_MINUTES
                )

                if is_expired:
                    redis_client.delete(key)
                    cleaned_count += 1
                    logger.debug(f"Cleaned session {key} (reason: {reason})")

            except Exception as e:
                logger.warning(f"Error checking session {key}: {e}")

        # Break if cursor returned to 0 (full scan complete)
        if cursor == 0:
            break

    logger.info(f"Session cleanup complete: {cleaned_count} sessions removed")

def start_cleanup_scheduler():
    """Start background scheduler for session cleanup."""
    # Run every 5 minutes
    schedule.every(5).minutes.do(cleanup_expired_sessions)

    logger.info("Session cleanup scheduler started (runs every 5 minutes)")

    while True:
        schedule.run_pending()
        time.sleep(60)  # Check every minute

if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    start_cleanup_scheduler()
```

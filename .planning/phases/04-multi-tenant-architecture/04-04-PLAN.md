```python
import pytest
from unittest.mock import patch

def test_tenant_isolation_basic(test_client, setup_test_tenants):
    """Verify data from tenant A doesn't leak into tenant B."""
    tenants = setup_test_tenants

    # Mock JWT to return tenant_a
    with patch('src.auth.jwt.decode_jwt') as mock_jwt:
        mock_jwt.return_value = {"sub": "user-123", "tenant_id": str(tenants["tenant_a"].id)}

        # Create finding in tenant A
        response = test_client.post(
            "/findings",
            json={"title": "Finding A", "severity": "HIGH"},
            headers={"Authorization": "Bearer fake-token"}
        )
        assert response.status_code in [200, 201]

    # Mock JWT to return tenant_b
    with patch('src.auth.jwt.decode_jwt') as mock_jwt:
        mock_jwt.return_value = {"sub": "user-456", "tenant_id": str(tenants["tenant_b"].id)}

        # Query findings in tenant B - should NOT see tenant A's finding
        response = test_client.get(
            "/findings",
            headers={"Authorization": "Bearer fake-token"}
        )
        assert response.status_code == 200
        findings = response.json()
        assert all(f["title"] != "Finding A" for f in findings), "Tenant isolation violated!"

def test_rbac_integration_preserved(test_client, setup_test_tenants):
    """Verify Phase 3 RBAC still works with tenant routing."""
    tenants = setup_test_tenants

    # Mock JWT for authenticated user without permissions
    with patch('src.auth.jwt.decode_jwt') as mock_jwt:
        mock_jwt.return_value = {"sub": "user-no-perms", "tenant_id": str(tenants["tenant_a"].id)}

        # Mock get_user_permissions to return no permissions
        with patch('src.rbac.dependencies.get_user_permissions') as mock_perms:
            mock_perms.return_value = set()  # No permissions

            # Attempt to access protected endpoint
            response = test_client.get(
                "/findings",
                headers={"Authorization": "Bearer fake-token"}
            )
            # Should be 403 (RBAC denies) not 401 (auth failed)
            assert response.status_code == 403

def test_tenant_provisioning_flow(test_client):
    """Verify POST /tenants creates schema and provisions correctly."""
    # Mock admin permissions
    with patch('src.rbac.dependencies.require_permissions') as mock_rbac:
        mock_rbac.return_value = None  # Admin has admin:manage

        response = test_client.post(
            "/tenants",
            json={"slug": "test-provision", "name": "Test Provision Org"},
            headers={"Authorization": "Bearer admin-token"}
        )

        assert response.status_code == 201
        assert response.json()["slug"] == "test-provision"

        # Verify tenant created
        from src.api.database import SessionLocal
        db = SessionLocal()
        tenant = db.query(Tenant).filter(Tenant.slug == "test-provision").first()
        assert tenant is not None
        assert tenant.is_provisioned or tenant.migration_status == "pending"
        db.close()

def test_migration_status_api(test_client, setup_test_tenants):
    """Verify migration status can be queried."""
    # Mock admin permissions
    with patch('src.rbac.dependencies.require_permissions') as mock_rbac:
        mock_rbac.return_value = None

        response = test_client.get(
            "/tenants",
            headers={"Authorization": "Bearer admin-token"}
        )

        assert response.status_code == 200
        tenants_list = response.json()
        assert any(t["migration_status"] == "current" for t in tenants_list)
```

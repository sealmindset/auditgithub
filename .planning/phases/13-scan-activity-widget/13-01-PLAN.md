### Task 1: Add Recent Scans Analytics Endpoint

Add new endpoint to `src/api/routers/analytics.py` that returns recent scans across all repositories.

**Location:** After `recent_findings` endpoint (around line 200)

**Add endpoint:**
```python
@router.get("/recent-scans")
async def get_recent_scans(
    limit: int = 10,
    db: Session = Depends(get_db)
):
    """Get recent scan runs across all repositories."""
    query = db.query(models.ScanRun).options(
        joinedload(models.ScanRun.repository)
    )

    # Apply org filter for multi-tenant
    query = apply_org_filter(query, models.ScanRun, db)

    # Order by created_at desc, limit results
    scans = query.order_by(models.ScanRun.created_at.desc()).limit(limit).all()

    return [
        {
            "id": str(scan.id),
            "repository_name": scan.repository.name if scan.repository else "Unknown",
            "scan_type": scan.scan_type,
            "status": scan.status,
            "findings_count": scan.findings_count or 0,
            "new_findings_count": scan.new_findings_count or 0,
            "created_at": scan.created_at.isoformat() if scan.created_at else None,
            "completed_at": scan.completed_at.isoformat() if scan.completed_at else None,
            "duration_seconds": scan.duration_seconds,
            "triggered_by": scan.triggered_by
        }
        for scan in scans
    ]
```

**Add import at top if needed:**
```python

```

### Task 2: Create ScanActivityWidget Component

Create new widget component at `src/web-ui/components/dashboard/ScanActivityWidget.tsx`.

**Component structure:**
```tsx
"use client"

import { Widget, WidgetHeader, WidgetContent, WidgetSkeleton, WidgetError } from "./Widget"
import { useWidgetData } from "@/hooks/useWidgetData"
import { Badge } from "@/components/ui/badge"
import { CheckCircle2, XCircle, Clock, Loader2, Play } from "lucide-react"
import { formatDistanceToNow } from "date-fns"

interface ScanActivity {
  id: string
  repository_name: string
  scan_type: "full" | "incremental" | "validation"
  status: "queued" | "running" | "completed" | "failed"
  findings_count: number
  new_findings_count: number
  created_at: string
  completed_at: string | null
  duration_seconds: number | null
  triggered_by: string
}

const statusConfig = {
  completed: { icon: CheckCircle2, color: "text-green-500", bg: "bg-green-500/10" },
  failed: { icon: XCircle, color: "text-red-500", bg: "bg-red-500/10" },
  running: { icon: Loader2, color: "text-blue-500", bg: "bg-blue-500/10", animate: true },
  queued: { icon: Clock, color: "text-yellow-500", bg: "bg-yellow-500/10" }
}

const scanTypeLabels = {
  full: "Full",
  incremental: "Incr",
  validation: "Valid"
}

function formatDuration(seconds: number | null): string {
  if (!seconds) return "-"
  if (seconds < 60) return `${seconds}s`
  if (seconds < 3600) return `${Math.floor(seconds / 60)}m`
  return `${Math.floor(seconds / 3600)}h ${Math.floor((seconds % 3600) / 60)}m`
}

export function ScanActivityWidget() {
  const { data, isLoading, error } = useWidgetData<ScanActivity[]>(
    "/analytics/recent-scans",
    { refreshInterval: 30000 }
  )

  if (isLoading) return <WidgetSkeleton />
  if (error) return <WidgetError message={error.message} />

  const scans = data || []

  return (
    <Widget>
      <WidgetHeader
        title="Recent Scans"
        description="Last 10 scan activities"
        icon={<Play className="h-4 w-4" />}
      />
      <WidgetContent>
        {scans.length === 0 ? (
          <div className="text-center py-8 text-muted-foreground">
            No scan activity yet
          </div>
        ) : (
          <div className="space-y-2">
            {scans.map((scan) => {
              const config = statusConfig[scan.status]
              const StatusIcon = config.icon
              return (
                <div
                  key={scan.id}
                  className={`flex items-center justify-between p-2 rounded-lg ${config.bg}`}
                >
                  <div className="flex items-center gap-3 min-w-0">
                    <StatusIcon
                      className={`h-4 w-4 flex-shrink-0 ${config.color} ${config.animate ? "animate-spin" : ""}`}
                    />
                    <div className="min-w-0">
                      <p className="text-sm font-medium truncate">
                        {scan.repository_name}
                      </p>
                      <p className="text-xs text-muted-foreground">
                        {formatDistanceToNow(new Date(scan.created_at), { addSuffix: true })}
                      </p>
                    </div>
                  </div>
                  <div className="flex items-center gap-2 flex-shrink-0">
                    <Badge variant="outline" className="text-xs">
                      {scanTypeLabels[scan.scan_type]}
                    </Badge>
                    {scan.status === "completed" && scan.new_findings_count > 0 && (
                      <Badge variant="destructive" className="text-xs">
                        +{scan.new_findings_count}
                      </Badge>
                    )}
                    <span className="text-xs text-muted-foreground w-12 text-right">
                      {formatDuration(scan.duration_seconds)}
                    </span>
                  </div>
                </div>
              )
            })}
          </div>
        )}
      </WidgetContent>
    </Widget>
  )
}
```

**Verification:**
- Component renders without errors
- Shows loading skeleton initially
- Displays scan list with status icons and badges
